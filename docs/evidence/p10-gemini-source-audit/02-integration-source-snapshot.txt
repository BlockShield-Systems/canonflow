### `services/mcp-clickhouse/scripts/verify_mcp.py`

```text
   1: import asyncio
   2: import os
   3: import uuid
   4: 
   5: from fastmcp import Client
   6: 
   7: 
   8: MCP_URL = "http://127.0.0.1:8000/mcp"
   9: PROJECT_SLUG = "yd-when-paradise-glitches"
  10: 
  11: 
  12: async def call(client: Client, tool: str, arguments: dict):
  13:     result = await client.call_tool(tool, arguments)
  14: 
  15:     if result.is_error:
  16:         raise RuntimeError(f"{tool} failed: {result}")
  17: 
  18:     return result
  19: 
  20: 
  21: async def main() -> None:
  22:     token = os.environ["CLICKHOUSE_MCP_AUTH_TOKEN"]
  23:     run_id = str(uuid.uuid4())
  24: 
  25:     async with Client(MCP_URL, auth=token) as client:
  26:         tools = await client.list_tools()
  27:         tool_names = sorted(tool.name for tool in tools)
  28: 
  29:         print("MCP tools:", tool_names)
  30: 
  31:         required_tools = {
  32:             "list_databases",
  33:             "list_tables",
  34:             "run_query",
  35:         }
  36: 
  37:         missing = required_tools.difference(tool_names)
  38:         if missing:
  39:             raise RuntimeError(f"Missing MCP tools: {sorted(missing)}")
  40: 
  41:         databases = await call(client, "list_databases", {})
  42:         print("Databases:", databases.data)
  43: 
  44:         tables = await call(
  45:             client,
  46:             "list_tables",
  47:             {
  48:                 "database": "canonflow",
  49:                 "include_detailed_columns": False,
  50:                 "page_size": 50,
  51:             },
  52:         )
  53:         print("Tables:", tables.data)
  54: 
  55:         project = await call(
  56:             client,
  57:             "run_query",
  58:             {
  59:                 "query": f"""
  60:                     SELECT
  61:                         project_id,
  62:                         slug,
  63:                         title,
  64:                         status
  65:                     FROM canonflow.projects FINAL
  66:                     WHERE slug = '{PROJECT_SLUG}'
  67:                     LIMIT 1
  68:                 """
  69:             },
  70:         )
  71:         print("Project:", project.data)
  72: 
  73:         inserted = await call(
  74:             client,
  75:             "run_query",
  76:             {
  77:                 "query": f"""
  78:                     INSERT INTO canonflow.agent_runs
  79:                     (
  80:                         run_id,
  81:                         project_id,
  82:                         agent_name,
  83:                         workflow_name,
  84:                         model_name,
  85:                         status,
  86:                         input_summary,
  87:                         output_summary,
  88:                         completed_at,
  89:                         duration_ms,
  90:                         metadata_json
  91:                     )
  92:                     SELECT
  93:                         toUUID('{run_id}'),
  94:                         project_id,
  95:                         'mcp-verification-agent',
  96:                         'verify-mcp-runtime',
  97:                         'none',
  98:                         'completed',
  99:                         'Verify MCP read and write access',
 100:                         'MCP runtime verification succeeded',
 101:                         now64(3),
 102:                         0,
 103:                         '{{"verification":true,"transport":"streamable-http"}}'
 104:                     FROM canonflow.projects FINAL
 105:                     WHERE slug = '{PROJECT_SLUG}'
 106:                 """
 107:             },
 108:         )
 109:         print("Insert result:", inserted.data)
 110: 
 111:         verification = await call(
 112:             client,
 113:             "run_query",
 114:             {
 115:                 "query": f"""
 116:                     SELECT
 117:                         run_id,
 118:                         agent_name,
 119:                         workflow_name,
 120:                         status,
 121:                         output_summary
 122:                     FROM canonflow.agent_runs
 123:                     WHERE run_id = toUUID('{run_id}')
 124:                 """
 125:             },
 126:         )
 127:         print("Written record:", verification.data)
 128: 
 129:         print("MCP runtime verification: SUCCESS")
```
